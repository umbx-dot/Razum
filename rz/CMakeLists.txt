file(GLOB_RECURSE RZ_SOURCES CONFIGURE_DEPENDS
    main.c cli/*.c manifest/*.c targets/*.c packages/*.c support/*.c)
list(FILTER RZ_SOURCES EXCLUDE REGEX "/support/(windows|linux)/")
file(GLOB RZ_HOST_SOURCES CONFIGURE_DEPENDS support/${RAZUM_HOST}/*.c)
set(RZ_SHARED_SOURCE
    arena_alloc arena_copy_text fatal_error list_push read_file write_file text_append text_append_bytes text_line text_append_formatted)
list(TRANSFORM RZ_SHARED_SOURCE PREPEND "${PROJECT_SOURCE_DIR}/src/common/")
list(TRANSFORM RZ_SHARED_SOURCE APPEND ".c")
set(RZ_PLATFORM_SOURCE find_program list_files make_executable path_kind process_id run_process)
list(TRANSFORM RZ_PLATFORM_SOURCE PREPEND "${PROJECT_SOURCE_DIR}/src/platform/${RAZUM_HOST}/")
list(TRANSFORM RZ_PLATFORM_SOURCE APPEND ".c")

add_library(razum-toml STATIC vendor/tomlc17/tomlc17.c)
if(MSVC)
    target_compile_definitions(razum-toml PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
add_executable(rz ${RZ_SOURCES} ${RZ_HOST_SOURCES} ${RZ_SHARED_SOURCE} ${RZ_PLATFORM_SOURCE})
add_dependencies(rz rzc)
target_link_libraries(rz PRIVATE razum-toml)
target_include_directories(rz PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${PROJECT_SOURCE_DIR}/src")
set_target_properties(rz PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}")
if(WIN32)
    set(manifest "${PROJECT_SOURCE_DIR}/src/platform/windows/rzc.rc")
    target_sources(rz PRIVATE "${manifest}")
    set_property(SOURCE "${manifest}" APPEND PROPERTY
                 OBJECT_DEPENDS "${PROJECT_SOURCE_DIR}/src/platform/windows/rzc.manifest")
    if(MSVC)
        target_link_options(rz PRIVATE /MANIFEST:NO)
    endif()
endif()
if(MSVC)
    target_compile_definitions(rz PRIVATE _CRT_SECURE_NO_WARNINGS)
    target_compile_options(rz PRIVATE /W4 /WX)
else()
    target_compile_options(rz PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
install(TARGETS rz RUNTIME DESTINATION bin)
install(FILES vendor/tomlc17/LICENSE DESTINATION share/razum RENAME tomlc17-LICENSE.txt)
